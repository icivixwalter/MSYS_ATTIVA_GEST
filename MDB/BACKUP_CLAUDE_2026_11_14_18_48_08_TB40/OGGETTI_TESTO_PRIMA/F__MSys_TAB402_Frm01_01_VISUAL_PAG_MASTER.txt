'## #02# = NOMEOGGETTO_s_Txt_BeforeUpdate = quando scrivi un nome form, controlla che la form esista
'##################################################################################################
Private Sub NOMEOGGETTO_s_Txt_BeforeUpdate(Cancel As Integer)
Const LOG_PROC_s As String = "NOMEOGGETTO_s_Txt_BeforeUpdate"
Dim sPar_s As String
Dim sNome_s As String
On Error GoTo Err_SEZ
    sPar_s = "02.00": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "ingresso procedura"

    '//#02.01# = campo vuoto: nessun controllo
    sPar_s = "02.01"
    sNome_s = Trim(Nz(Me.NOMEOGGETTO_s_Txt.Value, ""))
    If sNome_s = "" Then
        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", "campo NOMEOGGETTO_s vuoto: nessun controllo"
        GoTo Exit_SEZ
    End If

    '//#02.02# = verifico l'esistenza della form
    sPar_s = "02.02": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "verifico l'esistenza della form: " & sNome_s
    If MSys_FORM_ESISTE_pFunct(sNome_s) Then
        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "OK", "form esistente: " & sNome_s
        GoTo Exit_SEZ
    End If

    '//#02.03# = form inesistente: SI = salvo con VISIBLE_b=False / NO = annullo
    sPar_s = "02.03": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "AVVISO", "FORM NON ESISTE: " & sNome_s
    If MsgBox("La form '" & sNome_s & "' NON ESISTE nel database corrente." & vbCrLf & vbCrLf & _
              "SI  = salvo il nome ma imposto VISIBLE_b = False (la pagina non verra caricata)" & vbCrLf & _
              "NO = annullo la modifica e correggo il nome", _
              vbExclamation + vbYesNo, "CONTROLLO FORM #02.03#") = vbYes Then
        Me.VISIBLE_b_TXT.Value = False
        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "AVVISO", "salvato con VISIBLE_b=False: " & sNome_s
    Else
        Cancel = True
        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", "modifica annullata dall'utente: " & sNome_s
    End If

Exit_SEZ:
    LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "FINE", "uscita procedura"
    Exit Sub
Err_SEZ:
    LOG_Errore LOG_CLASSE_s, LOG_PROC_s, sPar_s, Err.Number, Err.Description
    MsgBox "ERRORE " & Err.Number & ": " & Err.Description & vbCrLf & vbCrLf & _
           "Sezione/Paragrafo: #" & sPar_s & "#  in  " & LOG_PROC_s & " (" & LOG_CLASSE_s & ")" & vbCrLf & _
           "(dettagli in Logs\LogEventi.txt)", vbCritical, "ERRORE #" & sPar_s & "#"
    Resume Exit_SEZ
End Sub

'##################################################################################################
'## #03# = VISIBLE_b_TXT_BeforeUpdate = impedisce di rendere visibile una pagina con form inesistente
'##################################################################################################
Private Sub VISIBLE_b_TXT_BeforeUpdate(Cancel As Integer)
Const LOG_PROC_s As String = "VISIBLE_b_TXT_BeforeUpdate"
Dim sPar_s As String
Dim sNome_s As String
On Error GoTo Err_SEZ
    sPar_s = "03.00": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "ingresso procedura"

    '//#03.01# = se tolgo la spunta: nessun controllo
    sPar_s = "03.01"
    If Nz(Me.VISIBLE_b_TXT.Value, False) = False Then GoTo Exit_SEZ

    '//#03.02# = verifico l'esistenza della form indicata nella riga
    sPar_s = "03.02"
    sNome_s = Trim(Nz(Me.NOMEOGGETTO_s_Txt.Value, ""))
    If MSys_FORM_ESISTE_pFunct(sNome_s) Then
        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "OK", "pagina resa visibile, form esistente: " & sNome_s
        GoTo Exit_SEZ
    End If

    '//#03.03# = form inesistente: annullo la modifica
    sPar_s = "03.03": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "AVVISO", "VISIBLE_b=True rifiutato, form inesistente: '" & sNome_s & "'"
    MsgBox "Non puoi rendere visibile questa pagina:" & vbCrLf & _
           "la form '" & sNome_s & "' NON ESISTE nel database corrente.", vbExclamation, "CONTROLLO FORM #03.03#"
    Cancel = True

Exit_SEZ:
    LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "FINE", "uscita procedura"
    Exit Sub
Err_SEZ:
    LOG_Errore LOG_CLASSE_s, LOG_PROC_s, sPar_s, Err.Number, Err.Description
    MsgBox "ERRORE " & Err.Number & ": " & Err.Description & vbCrLf & vbCrLf & _
           "Sezione/Paragrafo: #" & sPar_s & "#  in  " & LOG_PROC_s & " (" & LOG_CLASSE_s & ")" & vbCrLf & _
           "(dettagli in Logs\LogEventi.txt)", vbCritical, "ERRORE #" & sPar_s & "#"
    Resume Exit_SEZ
End Sub
