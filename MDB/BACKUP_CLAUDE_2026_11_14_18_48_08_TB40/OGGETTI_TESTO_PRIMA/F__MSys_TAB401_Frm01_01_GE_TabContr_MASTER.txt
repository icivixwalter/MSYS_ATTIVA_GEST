sPar_s = "06.00": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "ingresso procedura"

                Dim FormAperta_b As Boolean
                Dim Valore_b As Boolean

                                    sSql = sSql & "WHERE (((MSys_TAB401_Qry01_01_GE_TabContr_MASTER.Scel_b)=True));"
                                        
                                        Valore_b = False
                                
                                        Set DaoRs = CurrentDb.OpenRecordset(sSql)
                                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", sSql
                                                            
                                            With DaoRs
                                                
                                                    While Not DaoRs.EOF
                        
                                                    If .Fields("Scel_b") = True Then
                                                            Valore_b = True
                                                            Vv1 = .Fields("COD_PROGETTO_s")
                                                    End If
                                               
                                                   .MoveNext
                                                    Wend
                                                                        
                                            End With
                                                        
                                        DaoRs.Close
                                        Set DaoRs = Nothing
                                        
                                        LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", "Progetto con Scel_b=True trovato: " & Valore_b & " -> " & Vv1
                                        If Valore_b = False Then
                                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "AVVISO", "nessun progetto con Scel_b=True: uscita"
                                            MsgBox "ATTENZIONE DEVI IMPOSTARE UN PROGETTO!", vbCritical, "CONTROLLO PROGETTI IMPOSTATI"
                                            Me.Scel_b = True = True
                                            Me.Requery
                                            GoTo Exit_COD_PROGETTO_s_Txt_DblClick
                                        End If
                                        
'//---------------------------------------------------------------------------------------------------------//
'STEP 03 = Svuoto la tabella Control Master TMP e inserisco il progetto scelto

                                  '//#06.04# = ricreo MSys_TAB401_Qry01_01_GE_TabContr_MASTER_TMP (delete + Insert_toTMP)

                                  sSql = sSql & "DELETE MSys_TAB401_Qry01_01_GE_TabContr_MASTER_TMP.* FROM MSys_TAB401_Qry01_01_GE_TabContr_MASTER_TMP; "
                                               
                                  LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", sSql
                                  CurrentDb.Execute (sSql)
                                  LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "OK", "svuotata MSys_TAB401_Qry01_01_GE_TabContr_MASTER_TMP"
                                  
                                  DoCmd.OpenQuery "MSys_TAB401_Qry01_01_GE_TabContr_MASTER_Insert_toTMP"

'//---------------------------------------------------------------------------------------------------------//
                                '//#06.05# = svuoto MSys_TAB402_VISUAL_PAG_MASTER_TMP (query ..._Delete_TMP)

'STEP 05 = Inserisco le pagine collegate al progetto scelto nella tabella TMP

                                  '//#06.06# = riempio MSys_TAB402_VISUAL_PAG_MASTER_TMP (query ..._Insert_TMP)
                                  sPar_s = "06.06": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "riempio MSys_TAB402_VISUAL_PAG_MASTER_TMP (query ..._Insert_TMP)"

'//---------------------------------------------------------------------------------------------------------//

                '//#06.07# = cerco la form MASTER aperta su cui ricaricare le pagine
                sPar_s = "06.07": LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INIZIO", "cerco la form MASTER aperta su cui ricaricare le pagine"

                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", frmName_s
                            FormAperta_b = True
                    End If
                     
                    If frm.Name = "ANAG_FrmDF01_M01_GESTIONE" Then
                        frmName_s = frm.Name
                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", frmName_s

                            FormAperta_b = True
                    End If
                    
                    If frmName_s = Null Or frmName_s = "" Then
                        frmName_s = frm.Name
                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", frmName_s
                            FormAperta_b = True
                    End If
                            
                            'Debug.Print "--------------------------------------------------"   '//disattivato 2026-11-14: riga decorativa
                            'Debug.Print "               form corrente                      "   '//disattivato 2026-11-14: riga decorativa
                            LOG_Scrivi LOG_CLASSE_s, LOG_PROC_s, sPar_s, "INFO", "form aperta esaminata: " & frm.Name
                            'Debug.Print   '//disattivato 2026-11-14: riga vuota
                            'Debug.Print "--------------------------------------------------"   '//disattivato 2026-11-14: riga decorativa
                    
                                If FormAperta_b = True Then
                                
                                    For Each ctl In frm.Controls
                                        'Debug.Print ">>>"; ctl.Name   '//disattivato 2026-11-14: elenco di TUTTI i controlli: troppo voluminoso per il log
                                        If ctl.Name = "SottoForm_01" Then
                                        End If
                                    Next ctl
                                End If
                
                Next frm
                        
'//---------------------------------------------------------------------------------------------------------//
'STEP 07 = Se la form master è aperta eseguo Reset, Carica 12 pagine e Recalc
