' ============================================================
' @02 FUNZIONE Importa_LINK_TABELLE_PFunct
' SCOPO:
'   Collega tutte le tabelle definite in GetTabellaPaths.
'   - Controlla NomeTabella e PercorsoFile
'   - Scrive un log dettagliato
'
' MODIFICHE APPLICATE:
'   - Usa la Collection di Array come specificato (Item è Variant che contiene un Array)
'   - Estrae NomeTabella e PercorsoFile con CStr/Trim$ per sicurezza
'   - Migliore gestione di Err.Clear e On Error per ogni iterazione
' ============================================================
Public Function Importa_LINK_TABELLE_PFunct()
    On Error GoTo Importa_Err

    Dim TabellaPath As Collection
    Set TabellaPath = GetTabellaPaths_LINK_TABELLE()
    
    Dim NomeTabella As String
    Dim PercorsoFile As String
    Dim i As Long
    Dim LogFile As Integer
    Dim LogFileName As String
    Dim ConteggioOk As Long
    Dim ConteggioErr As Long
    Dim Item As Variant        ' ogni elemento è un Array -> deve essere Variant

    If TabellaPath Is Nothing Then
        MsgBox "Errore: impossibile ottenere la collection delle tabelle.", vbExclamation
        Exit Function
    End If

    If TabellaPath.Count = 0 Then
        MsgBox "La collection delle tabelle è vuota. Nessuna operazione effettuata.", vbInformation
        Exit Function
    End If

    ' Creazione file log
    LogFileName = CurrentProject.Path & "\CollegaTabelle_LINK_TABELLE_LLPP_ATTI_Tb02_Allegati_LOG.txt"
    LogFile = FreeFile
    Open LogFileName For Output As #LogFile

    Print #LogFile, "Log esecuzione funzione: CollegaTabelle_LINK_TABELLE_LLPP_ATTI_Tb02_Allegati"
    Print #LogFile, "Data/Ora: " & Now()
    Print #LogFile, "--------------------------------------------"

    ConteggioOk = 0
    ConteggioErr = 0

    ' Ciclo su tutte le tabelle
    For i = 1 To TabellaPath.Count
        Item = TabellaPath(i)     ' Item è Array: Item(0)=Nome, Item(1)=Percorso

        ' Estrazione sicura in variabili tipizzate
        NomeTabella = Trim$(CStr(Item(0) & ""))
        PercorsoFile = Trim$(CStr(Item(1) & ""))

        ' Controllo valori vuoti
        If Len(NomeTabella) = 0 Or Len(PercorsoFile) = 0 Then
            Print #LogFile, Now & " - SKIPPED: Nome tabella o percorso vuoto"
            ConteggioErr = ConteggioErr + 1
            GoTo ProssimaTabella_Importa
        End If

        ' Controllo percorso file
        If Not FileEsiste(PercorsoFile) Then
            Print #LogFile, Now & " - ERRORE: Percorso file NON valido per " & NomeTabella & " -> " & PercorsoFile
            ConteggioErr = ConteggioErr + 1
            GoTo ProssimaTabella_Importa
        End If

        ' Elimina tabella se già esiste, poi crea collegamento
        On Error Resume Next
        Err.Clear
        EliminaTabellaSeEsiste_LINK_TABELLE NomeTabella
        DoCmd.TransferDatabase acLink, "Microsoft Access", PercorsoFile, acTable, NomeTabella, NomeTabella, False

        If Err.Number = 0 Then
            Print #LogFile, Now & " - OK: " & NomeTabella & " -> " & PercorsoFile
            ConteggioOk = ConteggioOk + 1
        Else
            Print #LogFile, Now & " - ERRORE: " & NomeTabella & " -> " & Err.Description
            ConteggioErr = ConteggioErr + 1
            Err.Clear
        End If

ProssimaTabella_Importa:
        On Error GoTo 0
    Next i

    ' Riepilogo log
    Print #LogFile, "--------------------------------------------"
    Print #LogFile, "TOTALE TABELLE COLLEGATE: " & ConteggioOk
    Print #LogFile, "TOTALE ERRORI/SKIPPED: " & ConteggioErr
    Close #LogFile

    MsgBox "Operazione LINK TABELLE completata!" & vbCrLf & _
           "Tabelle collegate: " & ConteggioOk & vbCrLf & _
           "Errori/Skipped: " & ConteggioErr & vbCrLf & _
           "Log salvato in: " & LogFileName, vbInformation

Importa_LINK_TABELLE_PFunct_Exit:
    Exit Function

Importa_Err:
    MsgBox "Errore: " & Err.Description, vbExclamation, "Errore di collegamento tabelle"
    Resume Importa_LINK_TABELLE_PFunct_Exit
End Function



' ============================================================
' @04 FUNZIONE EliminaTabellaSeEsiste_LINK_TABELLE
' SCOPO:
'   Elimina una tabella dal database corrente se esiste
' NOTA:
'   Funzione identica all'originale — compatibile con la Collection di Array
' ============================================================
Public Sub EliminaTabellaSeEsiste_LINK_TABELLE(NomeTabella As Variant)
    On Error Resume Next
    Dim tdf As DAO.TableDef
    For Each tdf In CurrentDb.TableDefs
        If tdf.Name = NomeTabella Then
            CurrentDb.TableDefs.Delete NomeTabella
            Exit For
        End If
    Next tdf
    On Error GoTo 0
End Sub



' ============================================================
' @06 FUNZIONE CancellaTutto_INK_TABELLE
' SCOPO:
'   Cancella tutte le tabelle collegate definite in GetTabellaPaths.
'   - Scrive log dettagliato su file
'   - Controlla validità NomeTabella
'   - Verifica l’esistenza della tabella nel DB corrente e la elimina
'
' MODIFICHE APPLICATE:
'   - Usa la Collection contenente Array come richiesto (Item è Variant)
'   - Converte in stringa con CStr/Trim$ per evitare errori tipo Variant/Null
' ============================================================
Public Function CancellaTutto_INK_TABELLE()
    On Error GoTo CancellaTutto_INK_TABELLE_Err

    Dim TabellaPath As Collection
    Set TabellaPath = GetTabellaPaths_LINK_TABELLE()

    ' === Controllo collection vuota ===
    If TabellaPath Is Nothing Then
        MsgBox "Errore: impossibile ottenere la collection delle tabelle.", vbExclamation
        Exit Function
    End If

    If TabellaPath.Count = 0 Then
        MsgBox "La collection delle tabelle è vuota. Nessuna tabella da cancellare.", vbInformation
        Exit Function
    End If

    Dim i As Long
    Dim Item As Variant        ' ogni elemento è un Array -> deve essere Variant
    Dim NomeTabella As String
    Dim PercorsoFile As String
    Dim ConteggioOk As Long
    Dim ConteggioErr As Long
    Dim tdf As DAO.TableDef
    Dim TabellaEsiste As Boolean
    Dim LogFile As Integer
    Dim LogFileName As String

    ' === Inizializza log ===
    LogFileName = CurrentProject.Path & "\CANC_TutteTabelleCollegate_PFunct_LOG.txt"
    LogFile = FreeFile
    Open LogFileName For Output As #LogFile

    Print #LogFile, "Log esecuzione funzione: CancellaTutteTabelleCollegate_LINK_TABELLE"
    Print #LogFile, "Data/Ora: " & Now()
    Print #LogFile, "--------------------------------------------"

    ConteggioOk = 0
    ConteggioErr = 0

    ' === Ciclo sugli elementi della collection ===
    For i = 1 To TabellaPath.Count
        Item = TabellaPath(i) ' ogni elemento è un array con NomeTabella e PercorsoFile

        ' --- Estrazione sicura in variabili tipizzate ---
        NomeTabella = Trim$(CStr(Item(0) & ""))
        PercorsoFile = Trim$(CStr(Item(1) & ""))
        
        ' --- Controllo nome tabella vuoto ---
        If Len(NomeTabella) = 0 Then
            Print #LogFile, Now & " - SKIPPED: Nome tabella vuota"
            ConteggioErr = ConteggioErr + 1
            GoTo ProssimaTabella_Cancella
        End If

        ' --- Verifica se la tabella esiste nel DB corrente ---
        TabellaEsiste = False
        For Each tdf In CurrentDb.TableDefs
            If tdf.Name = NomeTabella Then
                TabellaEsiste = True
                Exit For
            End If
        Next tdf
            
        ' === Aggiorna campo Txt Form PATH DB  ===
        sourceDBPath_s_Txt = PercorsoFile
        
        If TabellaEsiste Then
            On Error Resume Next
            Err.Clear
            EliminaTabellaSeEsiste_LINK_TABELLE NomeTabella
            If Err.Number = 0 Then
                Print #LogFile, Now & " - OK: Eliminata " & NomeTabella
                ConteggioOk = ConteggioOk + 1
            Else
                Print #LogFile, Now & " - ERRORE: " & NomeTabella & " -> " & Err.Description
                ConteggioErr = ConteggioErr + 1
                Err.Clear
            End If
            On Error GoTo 0
        Else
            Print #LogFile, Now & " - NON TROVATA: " & NomeTabella
            ConteggioErr = ConteggioErr + 1
        End If

ProssimaTabella_Cancella:
    Next i

    ' === Riepilogo log ===
    Print #LogFile, "--------------------------------------------"
    Print #LogFile, "TOTALE TABELLE ELIMINATE: " & ConteggioOk
    Print #LogFile, "TOTALE ERRORI/SKIPPED: " & ConteggioErr
    Close #LogFile

    ' === aggiorna riquadro oggetti ===
    Call RefreshDatabaseView

    MsgBox "Cancellazione LINK TABELLE completata!" & vbCrLf & _
           "Tabelle eliminate: " & ConteggioOk & vbCrLf & _
           "Errori/Skipped: " & ConteggioErr & vbCrLf & _
           "Log salvato in: " & LogFileName, vbInformation

CancellaTutto_INK_TABELLE_Exit:
    Exit Function

CancellaTutto_INK_TABELLE_Err:
    MsgBox "Errore: " & Err.Description, vbExclamation, "Errore cancellazione tabelle"
    Resume CancellaTutto_INK_TABELLE_Exit
End Function

' ============================================================
' FUNZIONE Importa_QUERY_PFunct
' SCOPO:
'   Importa tutte le query definite in GetQueryPaths_QUERY
'   - Scrive log dettagliato
'
' MODIFICHE APPLICATE:
'   - Estrazione sicura Item(0)/Item(1) con Trim$(CStr(...))
'   - Verifiche FileEsiste prima di importare
'   - Gestione Err.Clear tra le iterazioni
' ============================================================
Public Function Importa_QUERY_PFunct()
    On Error GoTo ErrHandler

    Dim QueryPath As Collection
    Set QueryPath = GetQueryPaths_QUERY()
    
    Dim NomeQuery As String
    Dim PercorsoFile As String
    Dim i As Long
    Dim LogFile As Integer
    Dim LogFileName As String
    Dim ConteggioOk As Long
    Dim ConteggioErr As Long
    Dim Item As Variant
    
    If QueryPath Is Nothing Then
        MsgBox "Errore: impossibile ottenere la collection delle query.", vbExclamation
        Exit Function
    End If

    If QueryPath.Count = 0 Then
        MsgBox "Nessuna query da collegare.", vbInformation
        Exit Function
    End If
    
    ' File log (cartella CurrentProject.Path)
    LogFileName = CurrentProject.Path & "\CollegaQuery_LINK_TABELLE_LLPP_ATTI_Tb02_Allegati_LOG.txt"
    LogFile = FreeFile
    Open LogFileName For Output As #LogFile
    Print #LogFile, "Log esecuzione funzione: Importa_QUERY_PFunct"
    Print #LogFile, "Data/Ora: " & Now()
    Print #LogFile, "--------------------------------------------"

    ConteggioOk = 0
    ConteggioErr = 0

    ' Ciclo
    For i = 1 To QueryPath.Count
        Item = QueryPath(i)
        
        ' Estrazione sicura
        NomeQuery = Trim$(CStr(Item(0) & ""))
        PercorsoFile = Trim$(CStr(Item(1) & ""))
        
        ' Controllo NomeQuery
        If Len(NomeQuery) = 0 Then
            Print #LogFile, Now & " - SKIPPED: Nome query vuoto (indice " & i & ")"
            ConteggioErr = ConteggioErr + 1
            GoTo Prossimo
        End If
        
        ' Controllo file sorgente
        If Not FileEsiste(PercorsoFile) Then
            Print #LogFile, Now & " - ERRORE: File non trovato -> " & PercorsoFile & " (Query: " & NomeQuery & ")"
            ConteggioErr = ConteggioErr + 1
            GoTo Prossimo
        End If
        
        ' Elimina se già esiste (silenzioso, gestione errori dentro la routine)
        EliminaQuerySeEsiste_QUERY NomeQuery
        
        ' Collega / importa la query dal DB sorgente
        On Error Resume Next
        Err.Clear
        DoCmd.TransferDatabase _
            TransferType:=acImport, _
            DatabaseType:="Microsoft Access", _
            DatabaseName:=PercorsoFile, _
            objectType:=acQuery, _
            Source:=NomeQuery, _
            Destination:=NomeQuery, _
            StructureOnly:=False

        If Err.Number = 0 Then
            Print #LogFile, Now & " - OK: " & NomeQuery & " (Import da: " & PercorsoFile & ")"
            ConteggioOk = ConteggioOk + 1
        Else
            Print #LogFile, Now & " - ERRORE: " & NomeQuery & " -> " & Err.Description
            ConteggioErr = ConteggioErr + 1
            Err.Clear
        End If
        On Error GoTo 0

Prossimo:
    Next i
    
    ' Riepilogo
    Print #LogFile, "--------------------------------------------"
    Print #LogFile, "Totale collegate: " & ConteggioOk
    Print #LogFile, "Totale errori/skipped: " & ConteggioErr
    Close #LogFile
    
    ' Aggiorna vista oggetti (se presente)
    On Error Resume Next

' FUNZIONE CancellaTutte_QUERY
' SCOPO:

Public Function CancellaTutte_QUERY() As Boolean
    On Error GoTo ErrHandler
    
    Dim QueryPath As Collection

    If QueryPath Is Nothing Then
        MsgBox "Errore: impossibile ottenere la collection delle query.", vbExclamation
        CancellaTutte_QUERY = False
        Exit Function
    End If
    
    If QueryPath.Count = 0 Then
        MsgBox "Nessuna query da cancellare.", vbInformation
        CancellaTutte_QUERY = True
        Exit Function
    End If
    
    Dim i As Long, Item As Variant
    Dim NomeQuery As String
    Dim ConteggioOk As Long, ConteggioErr As Long

    Print #LogFile, "--------------------------------------------"
    
    ConteggioOk = 0
    ConteggioErr = 0
    
    For i = 1 To QueryPath.Count
        Item = QueryPath(i)
        NomeQuery = Trim$(CStr(Item(0) & ""))
        
        If Len(NomeQuery) = 0 Then
            Print #LogFile, Now & " - SKIPPED: Nome vuoto (indice " & i & ")"

                ConteggioOk = ConteggioOk + 1
            Else
                Print #LogFile, Now & " - ERRORE: Non eliminata " & NomeQuery

            End If
        Else

            ConteggioErr = ConteggioErr + 1
        End If
ProssimoCanc:
    Next i
    
    Print #LogFile, "--------------------------------------------"
    Print #LogFile, "Totale eliminate: " & ConteggioOk
    Print #LogFile, "Totale errori/skipped: " & ConteggioErr

    Call RefreshDatabaseView
    On Error GoTo 0

    MsgBox "Errore in CancellaTutte_QUERY: " & Err.Description, vbExclamation
    CancellaTutte_QUERY = False

' ============================================================
' SUB EliminaQuerySeEsiste_QUERY
' SCOPO:
'   Elimina una query dal database corrente se esiste
' ============================================================
Public Sub EliminaQuerySeEsiste_QUERY(NomeQuery As String)
    On Error Resume Next
    If Len(Trim$(NomeQuery)) > 0 Then
        CurrentDb.QueryDefs.Delete NomeQuery
    End If
    On Error GoTo 0
End Sub


' ============================================================
' FUNZIONE ESISTE_QUERY
' SCOPO:
'   Verifica se una query esiste nel DB corrente
' ============================================================
Public Function ESISTE_QUERY(NomeQuery As String) As Boolean
    Dim qdf As DAO.QueryDef
    ESISTE_QUERY = False
    On Error Resume Next
    For Each qdf In CurrentDb.QueryDefs
        If qdf.Name = NomeQuery Then
            ESISTE_QUERY = True
            Exit For
        End If
    Next qdf
    On Error GoTo 0
End Function

' ============================================================
' FUNZIONE Importa_FORM_PFunct
' SCOPO:
'   Collega tutte le form definite in GetFormPaths_FORM
'   - Scrive log dettagliato
' ============================================================
Public Function Importa_FORM_PFunct()
    On Error GoTo ErrHandler

    Dim FormPath As Collection
    Set FormPath = GetFormPaths_FORM()
    
    Dim NomeForm As String
    Dim PercorsoFile As String
    Dim i As Long
    Dim LogFile As Integer
    Dim LogFileName As String
    Dim ConteggioOk As Long
    Dim ConteggioErr As Long

    If FormPath.Count = 0 Then
        MsgBox "Nessuna form da collegare.", vbInformation
        Exit Function
    End If
    
    ' File log
    LogFileName = CurrentProject.Path & "\CollegaForm_LOG.txt"
    LogFile = FreeFile
    Open LogFileName For Output As #LogFile
    Print #LogFile, "Log esecuzione funzione: Importa_FORM_PFunct"
    Print #LogFile, "Data/Ora: " & Now()
    Print #LogFile, "--------------------------------------------"

    ' Ciclo
    For i = 1 To FormPath.Count
        Dim Item As Variant
        Item = FormPath(i)
        NomeForm = Item(0)
        PercorsoFile = Item(1)
        
        If Len(NomeForm) = 0 Then
            Print #LogFile, Now & " - SKIPPED: Nome form vuoto"
            ConteggioErr = ConteggioErr + 1
            GoTo Prossimo
        End If
        
        If Not FileEsiste(PercorsoFile) Then
            Print #LogFile, Now & " - ERRORE: File non trovato -> " & PercorsoFile
            ConteggioErr = ConteggioErr + 1
            GoTo Prossimo
        End If
        
        ' Elimina se già esiste
        EliminaFormSeEsiste_FORM NomeForm
        
        ' Collega (importa la Form dal DB sorgente)
        On Error Resume Next
        ' //@cmd@importa@FORM
        DoCmd.TransferDatabase _
            TransferType:=acImport, _
            DatabaseType:="Microsoft Access", _
            DatabaseName:=PercorsoFile, _
            objectType:=acForm, _
            Source:=NomeForm, _
            Destination:=NomeForm, _
            StructureOnly:=False

        If Err.Number = 0 Then
            Print #LogFile, Now & " - OK: " & NomeForm
            ConteggioOk = ConteggioOk + 1
        Else
            Print #LogFile, Now & " - ERRORE: " & NomeForm & " -> " & Err.Description
            ConteggioErr = ConteggioErr + 1
            Err.Clear
        End If
        On Error GoTo 0
Prossimo:
    Next i
    
    ' Riepilogo
    Print #LogFile, "--------------------------------------------"
    Print #LogFile, "Totale collegate: " & ConteggioOk
    Print #LogFile, "Totale errori/skipped: " & ConteggioErr
    Close #LogFile
    
    ' === aggiorna riquadro oggetti ===
    Call RefreshDatabaseView

    MsgBox "Operazione IMPORTA FORM completata!" & vbCrLf & _
           "Collegate: " & ConteggioOk & vbCrLf & _
           "Errori/Skipped: " & ConteggioErr & vbCrLf & _
           "Log: " & LogFileName, vbInformation
    Exit Function

ErrHandler:
    MsgBox "Errore in Importa_FORM_PFunct: " & Err.Description, vbExclamation
End Function

' @05 FUNZIONE FileEsiste
' SCOPO:
'   Funzione di utilità che controlla se un file esiste.
'   Restituisce TRUE se esiste, FALSE altrimenti.
' ============================================================
Public Function FileEsiste(PercorsoFile As String) As Boolean
    On Error Resume Next
    FileEsiste = (Dir(PercorsoFile, vbNormal) <> "")
    On Error GoTo 0
End Function
