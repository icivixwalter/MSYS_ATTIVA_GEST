'##################################################################################################
'## #01# = MSys_FORM_ESISTE_pFunct = restituisce True se la form esiste nel database corrente
'##        Usata da: MASTER #18.06# (controllo preventivo) e form 402 #02.02# / #03.02#
'##################################################################################################
Public Function MSys_FORM_ESISTE_pFunct(ByVal par_NomeForm_s As String) As Boolean
Dim obj As AccessObject
    '//#01.01# = cerco la form per nome (errore = non esiste)
    MSys_FORM_ESISTE_pFunct = False
    If Len(Trim(par_NomeForm_s)) = 0 Then Exit Function
    On Error Resume Next
    Set obj = CurrentProject.AllForms(Trim(par_NomeForm_s))
    MSys_FORM_ESISTE_pFunct = Not (obj Is Nothing)
    Err.Clear
    Set obj = Nothing
End Function
