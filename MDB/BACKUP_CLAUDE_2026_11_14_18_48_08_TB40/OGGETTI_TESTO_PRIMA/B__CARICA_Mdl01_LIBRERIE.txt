'2025_10_28_MODULO_LIBRERIE = creato un modulo per l'inserimento automatico delle librerie.
Sub ImpostaRiferimenti()
    Dim ref As Reference
    Dim refList As Variant
    Dim i As Integer
    Dim refPath As String
    Dim refName As String
    Dim giàPresente As Boolean

    ' Array con nome e percorso delle librerie
    refList = Array( _
        Array("VBE6", "C:\Program Files (x86)\Common Files\Microsoft Shared\VBA\VBA6\VBE6.DLL"), _
        Array("Access", "C:\Program Files (x86)\Microsoft Office\Office\MSACC9.OLB"), _
        Array("Office", "C:\Program Files (x86)\Microsoft Office\Office\MSO9.DLL"), _
        Array("ADODB", "C:\Program Files (x86)\Common Files\System\ado\msado21.tlb"), _
        Array("DAO", "C:\Program Files (x86)\Common Files\Microsoft Shared\DAO\dao360.dll") _
    )

    ' Scorre la lista e aggiunge se mancante
    For i = LBound(refList) To UBound(refList)
        refName = refList(i)(0)
        refPath = refList(i)(1)
        giàPresente = False

        ' Controlla se il riferimento esiste già
        For Each ref In Application.References
            If InStr(1, ref.fullPath, refName, vbTextCompare) > 0 Then
                giàPresente = True
                Exit For
            End If
        Next ref

        ' Se mancante, lo aggiunge
        If Not giàPresente Then
            On Error Resume Next
            Application.References.AddFromFile refPath
            If Err.Number = 0 Then
                Debug.Print "Aggiunto riferimento: " & refName
            Else
                Debug.Print "Errore aggiungendo " & refName & ": " & Err.Description
                Err.Clear
            End If
            On Error GoTo 0
        Else
            Debug.Print "Già presente: " & refName
        End If
    Next i

    MsgBox "Verifica completata. Controlla la finestra Immediata (Ctrl+G).", vbInformation
End Sub
