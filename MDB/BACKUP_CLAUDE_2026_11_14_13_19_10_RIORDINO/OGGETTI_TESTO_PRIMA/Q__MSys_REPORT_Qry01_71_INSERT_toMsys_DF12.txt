INSERT INTO MSys_ELENCO_ObjProgetti ( Scel_b, Name1_s, IdOrd_lng, NRO_OGGETTO_i, TIPOGGETTO_s, COD_PROGETTO_s, PROGETTO_s, DATAINS_d, TIMEOPER_d )
SELECT [MSys_REPORT].[Scel_b], [MSys_REPORT].[Name1_s], [MSys_REPORT].[ID_RPT_lng], [MSys_REPORT].[NRO_OGGETTO_i], "REPORTS" AS TIPOGGETTO_s, "MSys" AS COD_PROGETTO_s, "PROGETTO MSys" AS PROGETTO_s, [MSys_REPORT].[DATAINS_d], [MSys_REPORT].[TIMEOPER_d]
FROM MSys_REPORT LEFT JOIN MSys_ELENCO_ObjProgetti ON [MSys_REPORT].[Name1_s]=[MSys_ELENCO_ObjProgetti].[Name1_s]
WHERE ((([MSys_REPORT].[Scel_b])=True) And (([MSys_ELENCO_ObjProgetti].[Name1_s]) Is Null))
WITH OWNERACCESS OPTION;
