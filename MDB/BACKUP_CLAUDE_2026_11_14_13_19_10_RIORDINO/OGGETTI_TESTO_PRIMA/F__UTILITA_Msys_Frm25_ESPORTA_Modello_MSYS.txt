' ============================================================
' @01 FUNZIONE GetPaths_ESPORTAZIONI_Sub
' Restituisce la collection di tabelle e percorsi MDB da collegare
' Non vengono salvate nella collection le riga nulle.
' ============================================================
' COLLECTION MODULARE
' Un gruppo rappresenta un database di destinazione, ogni gruppo contiene una o più form da esportare.
' In VBA possiamo farlo in due modi:
'    Collection annidata (Collection di Collection)
'    Array dinamico contenente più form per ogni database
'
'    Soluzione 1: Collection annidata
'       Ogni elemento della collection principale contiene:
'       Percorso del database
'       Una collection secondaria con le form da esportare
'
'
' Vantaggi
'
'    Ogni database di destinazione ha la sua lista di form.
'    Puoi aggiungere form solo dentro la FormList senza duplicare righe di codice.
'    Il codice di esportazione diventa più pulito e scalabile.
' ============================================================



Private Function GetPaths_ESPORTAZIONI_Sub() As Collection
    On Error GoTo GestErrore
    
    ' ===========================================================
    ' Funzione: GetPaths_ESPORTAZIONI_Sub
    ' Scopo   : Restituisce una collection di "gruppi"
    '           - Ogni gruppo contiene:
    '               1. Percorso del database di destinazione
    '               2. Collection con l'elenco delle form da esportare
    ' ===========================================================
    
    Dim DbGroups As New Collection          ' Collection principale: contiene tutti i gruppi
    Dim FormList As Collection              ' Collection secondaria: contiene le form di ogni gruppo
    Dim PercorsoDatabase As String
    
    ' === Gruppo 1: Database LLPP_ATTI_GE_ALLEGATI.mdb ===
    PercorsoDatabase = "y:\LAVORI_PUBBLICI_FINO_AL_30_09_2019\Valter\OPERE_PUBBLICHE\96_ARCHIVI_LLPP_MDB\25_GESTIONE_LLPP_Y\ARCHIVI_ACCESS_LLPP_Y\LLPP_ATTI_Tb01_GESTIONE\MDB\GE_LLPP_ATTI_(Ricostruzione).mdb"
    If Dir(PercorsoDatabase) <> "" Then
        Set FormList = New Collection
        FormList.Add "UTILITA_Msys_Frm22_}-------------------------------------------@"
        FormList.Add "UTILITA_Msys_Frm22_CANCELLA_IMPORTA_LLPP_ATTI_Tb01_Gestione"
        
        FormList.Add "UTILITA_Msys_Frm25_}-------------------------------------------@"
        FormList.Add "UTILITA_Msys_Frm25_ESPORTA_Modello_MSYS"
        
        
        DbGroups.Add Array(PercorsoDatabase, FormList)
    End If
    
    ' === Gruppo 2: Database GE_DF01_GESTIONE.mdb===
    PercorsoDatabase = "y:\LAVORI_PUBBLICI_FINO_AL_30_09_2019\Valter\OPERE_PUBBLICHE\96_ARCHIVI_LLPP_MDB\25_GESTIONE_LLPP_Y\ARCHIVI_ACCESS_LLPP_Y\LLPP_DF01_CodiceOpera\MDB\GE_DF01_GESTIONE.mdb"
    If Dir(PercorsoDatabase) <> "" Then
        Set FormList = New Collection
        FormList.Add "UTILITA_Msys_Frm22_}-------------------------------------------@"
        FormList.Add "UTILITA_Msys_Frm22_CANCELLA_IMPORTA_LLPP_ATTI_Tb01_Gestione"
        
        
        FormList.Add "UTILITA_Msys_Frm25_}-------------------------------------------@"
        FormList.Add "UTILITA_Msys_Frm25_ESPORTA_Modello_MSYS"
        
        
        DbGroups.Add Array(PercorsoDatabase, FormList)
    End If
    
   
        
     
    
    
  
    ' Restituisce la collection principale con tutti i gruppi
    Set GetPaths_ESPORTAZIONI_Sub = DbGroups
    Exit Function

GestErrore:
    ' Gestione errori: restituisce Nothing in caso di errore
    MsgBox "Errore in GetPaths_ESPORTAZIONI_Sub: " & Err.Number & " - " & Err.Description, vbExclamation, "Errore"
    Set GetPaths_ESPORTAZIONI_Sub = Nothing
End Function

    

' ============================================================
' @02 FUNZIONE CollegaTabelle_LLPP_ATTI_Tb02_PFunct
' Collega le tabelle del database corrente ai MDB esterni
' ============================================================
Public Function EsportaForms_LLPP_PFunct()
    On Error GoTo EsportaForms_LLPP_PFunct_Err

    ' ===========================================================
    ' Funzione: EsportaForms_LLPP_PFunct
    ' Scopo   : Esporta una o più form dal database corrente verso
    '           uno o più database esterni (MDB), in base alla
    '           configurazione restituita da GetPaths_ESPORTAZIONI_Sub.
    ' Struttura:
    '   - Ciclo sui gruppi (uno per ogni database di destinazione)
    '   - Ciclo interno sulle form associate a quel database
    ' ===========================================================

    Dim DbGroups As Collection
    Set DbGroups = GetPaths_ESPORTAZIONI_Sub()  ' Ottiene la lista di database e relative form
    
    Dim PercorsoDbDest As String
    Dim FormList As Collection
    Dim NomeForm As String
    
    Dim i As Long, j As Long
    Dim LogFile As Integer
    Dim LogFileName As String
    Dim ConteggioOk As Long
    Dim ConteggioErr As Long

    ' ============================
    ' Controllo: nessun gruppo
    ' ============================
    If DbGroups.Count = 0 Then
        MsgBox "La collection dei database è vuota. Nessuna operazione effettuata.", vbInformation
        Exit Function
    End If

    ' ============================
    ' Apertura file di log
    ' ============================
    LogFileName = CurrentProject.Path & "\Msys_DF01_Mdl01_ESPORTA_FORM_LOG.txt"
    LogFile = FreeFile
    Open LogFileName For Output As #LogFile

    Print #LogFile, "Log esecuzione funzione: Msys_DF01_Mdl01_ESPORTA_FORM"
    Print #LogFile, "Data/Ora: " & Now()
    Print #LogFile, "--------------------------------------------"

    ConteggioOk = 0
    ConteggioErr = 0

    ' ============================
    ' Ciclo principale: database
    ' ============================
    For i = 1 To DbGroups.Count
        Dim Item As Variant
        Item = DbGroups(i)                  ' Array: [0] = percorso db, [1] = collection form
        PercorsoDbDest = Item(0)            ' Percorso del database di destinazione
        Set FormList = Item(1)              ' Elenco delle form da esportare

        Print #LogFile, Now & " === INIZIO ESPORTAZIONE SU: " & PercorsoDbDest

        ' ============================
        ' Ciclo interno: form
        ' ============================
        For j = 1 To FormList.Count
            NomeForm = FormList(j)          ' Nome della form da esportare

            ' Controllo valori validi
            If Len(NomeForm) = 0 Or Len(PercorsoDbDest) = 0 Then
                Print #LogFile, Now & " - SKIPPED: Nome form o percorso vuoto"
                ConteggioErr = ConteggioErr + 1
                GoTo ProssimaForm
            End If

            ' Tentativo di esportazione
            On Error Resume Next
            DoCmd.TransferDatabase _
                TransferType:=acExport, _
                DatabaseType:="Microsoft Access", _
                DatabaseName:=PercorsoDbDest, _
                objectType:=acForm, _
                Source:=NomeForm, _
                Destination:=NomeForm, _
                StructureOnly:=False

            ' Verifica esito
            If Err.Number = 0 Then
                Print #LogFile, Now & " - OK: " & NomeForm & " -> " & PercorsoDbDest
                ConteggioOk = ConteggioOk + 1
            Else
                Print #LogFile, Now & " - ERRORE: " & NomeForm & " -> " & Err.Description
                ConteggioErr = ConteggioErr + 1
                Err.Clear
            End If
ProssimaForm:
            On Error GoTo 0
        Next j

        Print #LogFile, Now & " === FINE ESPORTAZIONE SU: " & PercorsoDbDest
        Print #LogFile, "--------------------------------------------"
    Next i

    ' ============================
    ' Scrittura riepilogo finale
    ' ============================
    Print #LogFile, "TOTALE FORM ESPORTATE: " & ConteggioOk
    Print #LogFile, "TOTALE FORM NON ESPORTATE/ERRATE: " & ConteggioErr
    Close #LogFile

    ' ============================
    ' Messaggio a video
    ' ============================
    MsgBox "Operazione completata!" & vbCrLf & _
           "Form esportate: " & ConteggioOk & vbCrLf & _
           "Errori/Skipped: " & ConteggioErr & vbCrLf & _
           "Log salvato in: " & LogFileName, vbInformation

EsportaForms_LLPP_PFunct_Exit:
    Exit Function

EsportaForms_LLPP_PFunct_Err:
    ' In caso di errore chiudo il log se rimasto aperto
    If LogFile > 0 Then
        Close #LogFile
    End If
    MsgBox "Errore: " & Err.Description, vbExclamation, "Errore di esportazione form"
    Resume EsportaForms_LLPP_PFunct_Exit
End Function
