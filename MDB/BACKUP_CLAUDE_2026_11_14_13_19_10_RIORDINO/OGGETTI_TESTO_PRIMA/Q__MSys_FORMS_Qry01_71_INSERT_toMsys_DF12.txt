INSERT INTO MSys_ELENCO_ObjProgetti ( Scel_b, Name1_s, IdOrd_lng, NRO_OGGETTO_i, TIPOGGETTO_s, COD_PROGETTO_s, PROGETTO_s, DATAINS_d, TIMEOPER_d )
SELECT MSys_FORMS.Scel_b, MSys_FORMS.Name1_s, MSys_FORMS.ID_FRM_lng, MSys_FORMS.NRO_OGGETTO_i, "FORMS" AS TIPOGGETTO_s, "MSys" AS COD_PROGETTO_s, "PROGETTO MSys" AS PROGETTO_s, MSys_FORMS.DATAINS_d, MSys_FORMS.TIMEOPER_d
FROM MSys_FORMS LEFT JOIN MSys_ELENCO_ObjProgetti ON MSys_FORMS.[Name1_s] = MSys_ELENCO_ObjProgetti.[Name1_s]
WHERE (((MSys_FORMS.Scel_b)=True) AND ((MSys_ELENCO_ObjProgetti.Name1_s) Is Null))
WITH OWNERACCESS OPTION;
